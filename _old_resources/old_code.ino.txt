void loop() {
    safetyFunc();
    
    if(calibrate == true && run == false) {
        int tempPos1 = savedPos.pos2;
        int tempPos2 = currentPos;
        int tempDist = tempPos1 - tempPos2;
        nextPos = (tempDist / 5) + currentPos;
        timer.start();
        run = true;
    }
    if(gotoHome == true) {
        stopButton();
        programModeRunSpeed = 45;
        move();
        if(dataArray[HOMEpos] == 0) {
            Serial.println("Arrived at Home location.");
            stopTimer.start();
            stopCounter = 0;
            currentPos = 0;
            programModeRunSpeed = 0;
            gotoHome = false;
            
        }
    }else{
        buttonInputs();
        if(programMode == true) {
            move();
        }
    }
    
    if(checkForHomingDelay == true) {
        if(buttonState4 == HIGH) {
            if(millis() - homingDelayTime > 3000) {
                gotoHome = true;
                relay(0, 0);
                checkForHomingDelay = false;
            }
        }
    }
    
    if (stringComplete) {
        for(int i = 0; i < inputString.length(); i++) {
            dataArray[i] = inputString[i] - 48;
        }
        Serial.print("Received data: ");
        Serial.println(inputString);
        dataFunc();
        
        short sPinState1 = (short)dataArray[0];
        short sPinState2 = (short)dataArray[1];
        short sPinState3 = (short)dataArray[2];
        tempSpeedFactor = sPinState1 + (sPinState2 * 2) + (sPinState3 * 4);
        if(lastTempSpeedFactor != tempSpeedFactor) {
            MAXSPEED = (tempSpeedFactor * 15) + 15;
            Serial.printf("MAXSPEED set to: %d", MAXSPEED);
            lastTempSpeedFactor = tempSpeedFactor;
        }
        programMode = dataArray[3];
        if(dataArray[4] == true) {
            digitalWrite(RFLine, HIGH);
        }else {
            digitalWrite(RFLine, LOW);
        }
        if(lastprogState != dataArray[PROGpos]) {
            ProgTimer.start();
        }
        lastprogState = dataArray[PROGpos];
        // clear the string:
        inputString = "";
        stringComplete = false;
    }
    
    if(blinkValue > 0) {
        if(!(onWait.isActive()) && blinkLightOn == true) {
            relay(1, 1);
            onWait.start();
        }
        if(!(offWait.isActive()) && blinkLightOn == false) {
            relay(1, 0);
            offWait.start();
        }
    }
    
}

void safetyFunc() {
    unsigned long tempTime = millis();
    safetyState = digitalRead(A1);
    if(safetyState != lastSafetyState) {
        if(safetyState == true) {//3.3v means safety loop broken //&& arriveAtFloor == false
            callStop();
        }
    }
    lastSafetyState = safetyState;
    if(safetyState == false) {// low means safety loop closed
        if(tempTime - safetyTime > 750) { //run == false
            Serial1.print("(513)>:");
            safetyTime = tempTime;
        }
    }
}

void calibrateFunc() {
    unsigned short newThresh = 0;
    short tempNum = currentPos - nextPos;
    newThresh = abs(tempNum);
    savedPos.thresholdData[tempSpeedFactor] = newThresh;
    EEPROM.put(0, savedPos);
    Serial.printf("Threshold value calibrated to: %d", newThresh);
}

void stopButton() {

void gotoNextPos() {
    
    if(calculate == true) {
        lastPos = currentPos;
        
        totalDistance = nextPos - lastPos;
        totalDistance = abs(totalDistance);
        currDistance = 0;
        delayTimer.start();
        if(calibrate == false) {
            THRESHOLD = savedPos.thresholdData[tempSpeedFactor];
        }else {
            THRESHOLD = 0;
        }
        if(nextPos - lastPos >= 0) {
            neg = 1;
            threshStopPos = nextPos + THRESHOLD;
        }else{
            neg = -1;
            threshStopPos = nextPos - THRESHOLD;
        }
        calculate = false;
        speed = MAXSPEED;
        
    }
    currDistance = currentPos - lastPos;
    currDistance = abs(currDistance);
    if((currDistance >= totalDistance / 2 && reachedTime == false) || currDistance >= totalDistance - THRESHOLD || currentPos == threshStopPos) {
        Serial.printf("Threshold reached or short exception at pos %d\n", currentPos);
        if(calibrate == true) {
            calibrateTimer.start();
            calibrate = false;
        }
        //arriveAtFloor = true;
        callStop();
        relay(toFloorNum, 1);
        toFloorNum = -1;
    }else { //else if(confirmation == false)
        char data[8];
        String format = (neg < 0) ? "(2%s)" : "(1%s)";
        int speed2 = speed % 100;
        char setting[5] = { ((int) speed / 100) + 48, ((int) speed2 / 10) + 48, ((int) speed2 % 10) + 48, '0' };
        sprintf(data, format, setting);
        char checkSum = 0;
        for(int i = 0; i < 8; i++){
            checkSum += data[i];
        }
        checkSum = checkSum & 255;
        char sum1 = (checkSum / 0x10) + 0x30;
        char sum2 = (checkSum % 0x10) + 0x30;
        char *formatTwo = "%s%c%c";
        char finalData[10];
        sprintf(finalData, formatTwo, data, sum1, sum2);

        Serial1.print(finalData);
    }
    Serial.printf("Speed: %d, Position: %d, Target: %d\n", speed, currentPos, nextPos);
    
}

void buttonInputs() {
    // Left Button
  buttonState0 = digitalRead(Left);
  if (buttonState0 != lastButtonState0) {
    if (buttonState0 == HIGH && run == false) {
        relay(0, 0);
        delay(300);
        safetyState = digitalRead(A1);
      if(programMode == false && safetyState == false) {
        gotoSavedPos(2);
        Serial.println("Going to saved pos 2.");
      }
      Serial.println("Left button ON");
    }
  }
  lastButtonState0 = buttonState0;
  // Top Button
  buttonState3 = digitalRead(Top);
  if (buttonState3 != lastButtonState3) {
    if (buttonState3 == HIGH && run == false) {
        relay(0, 0);
        delay(300);
        safetyState = digitalRead(A1);
      if(programMode == true) {
        programModeRunSpeed = 60;
      }else if(safetyState == false){
        gotoSavedPos(3);
        Serial.println("Going to saved pos 3.");
        }
      Serial.println("Top button ON");
    }else{
        if(programMode == true) {
            programModeRunSpeed = 0;
        }
    }
  }
  lastButtonState3 = buttonState3;
  // Right Button
  buttonState2 = digitalRead(Right);
  if (buttonState2 != lastButtonState2) {
    if (buttonState2 == HIGH && run == false) {
        relay(0, 0);
        delay(300);
        safetyState = digitalRead(A1);
      if(programMode == false && safetyState == false) {
        gotoSavedPos(4);
        Serial.println("Going to saved pos 4.");
      }else if(programMode == true) {
        if(currRelayValue & 0x02 == 0x02) {
            relay(2, 0);
        }else{
            relay(2, 1);
        }
      }
      Serial.println("Right button ON");
    }
  }
  lastButtonState2 = buttonState2;
  // Center Button
  buttonState4 = digitalRead(Center);
  if (buttonState4 != lastButtonState4) {
    if (buttonState4 == HIGH) {
        checkForHomingDelay = true;
        homingDelayTime = millis();
        if(programMode == true) {
            setPos();
        }else{
            callStop();
        }
      Serial.println("Center button ON");
    }else{
        checkForHomingDelay = false;
    }
  }
  lastButtonState4 = buttonState4;
  
  // Bottom Button
  buttonState1 = digitalRead(Bottom);
  if (buttonState1 != lastButtonState1) {
    if (buttonState1 == HIGH) {
      if(programMode == true && run == false && safetyState == false) {
        programModeRunSpeed = -60;
      }else{
        if(currRelayValue & 0x01 == 0x01) {
            relay(1, 0);
        }else{
            relay(1, 1);
        }
      }
      Serial.println("Bottom button ON");
    }else{
        if(programMode == true) {
            programModeRunSpeed = 0;
        }
    }
  }
  lastButtonState1 = buttonState1;
}

void callStop() {
    gotoHome = false;
    calibrate = false;
    programModeRunSpeed = 0;
    speed = 0;
    stopTimer.start();
    stopCounter = 0;
    timer.stop();
    calculate = true;
    run = false;
}

void relay(uint8_t relayNum, uint8_t state) {
    switch(relayNum) {
        case 0: {
            if(state == 1) {
                currRelayValue = currRelayValue | 0x0F; 
            }else{
                currRelayValue = currRelayValue & 0xF0;
            }
            break;
        }
        case 1: {
            if(state == 1) {
                currRelayValue = currRelayValue | 0x01; 
            }else{
                currRelayValue = currRelayValue & 0xFE;
            }
            break;
        }
        case 2: {
            if(state == 1) {
                currRelayValue = currRelayValue | 0x02; 
            }else{
                currRelayValue = currRelayValue & 0xFD;
            }
            break;
        }
        case 3: {
            if(state == 1) {
                currRelayValue = currRelayValue | 0x04; 
            }else{
                currRelayValue = currRelayValue & 0xFB; 
            }
            break;
        }
        case 4: {
            if(state == 1) {
                currRelayValue = currRelayValue | 0x08; 
            }else{
                currRelayValue = currRelayValue & 0xF7;
            }
            break;
        }
        default: {
            break;
        }
    }
    relayMsg = "(";
    char tempChar = (currRelayValue & 0x01) + 48;
    relayMsg += tempChar;
    tempChar = ((currRelayValue & 0x02)>>1) + 48;
    relayMsg += tempChar;
    tempChar = ((currRelayValue & 0x04)>>2) + 48;
    relayMsg += tempChar;
    tempChar = ((currRelayValue & 0x08)>>3) + 48;
    relayMsg += tempChar;
    
    relayMsg += ")\n";
    Serial2.print(relayMsg);
    Serial.print("Sent: ");
    Serial.println(relayMsg);
    Serial.println(currRelayValue);
}
